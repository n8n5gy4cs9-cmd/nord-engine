resolveAttributes

Merge scene-provided attribute values over the class's static attributes defaults, validating types, clamping min/max.

Import from nord. Definition from runtime/core/script.js; names and defaults below are taken directly from the runtime.

export function resolveAttributes(Class, given = {}, warn = console.warn) {
  const out = {};
  const defs = Class?.attributes ?? {};
  for (const [k, def] of Object.entries(defs)) {
    const d = structuredClone(def.default);
    let v = k in given ? given[k] : d;
    if (def.type === 'enum') v = def.options?.includes(v) ? v : (def.options?.includes(d) ? d : def.options?.[0]);
    else if (def.type === 'array') v = Array.isArray(v) ? v : d ?? [];
    else if (def.type === 'curve') v = v ?? d;
    else if (coerce[def.type]) {
      const c = coerce[def.type](v, d);
      if (c !== v && k in given) warn(`[nord] attribute '${k}' expected ${def.type}, got ${JSON.stringify(v)}; using default`);
      v = c;
    }
    if (typeof v === 'number') {
      if (def.min != null) v = Math.max(def.min, v);
      if (def.max != null) v = Math.min(def.max, v);
    }
    out[k] = v;
  }
  for (const k of Object.keys(given)) if (!(k in defs)) warn(`[nord] unknown attribute '${k}' on ${Class?.name ?? 'script'}`);
  return out;
}

applyOverrides

Deep-merge prefab overrides onto entity data. Plain objects merge; objects with numeric keys patch arrays by index.

Import from nord. Definition from runtime/core/scene.js; names and defaults below are taken directly from the runtime.

export function applyOverrides(target, ov) {
  if (ov == null || typeof ov !== 'object') return ov;
  if (Array.isArray(target) && !Array.isArray(ov) && Object.keys(ov).every((k) => /^\d+$/.test(k))) {
    for (const [i, v] of Object.entries(ov)) target[i] = typeof v === 'object' && v !== null && !Array.isArray(v) ? applyOverrides(target[i] ?? {}, v) : v;
    return target;
  }
  if (Array.isArray(ov)) return structuredClone(ov);
  for (const [k, v] of Object.entries(ov)) {
    target[k] = v !== null && typeof v === 'object' && !Array.isArray(v) && typeof target[k] === 'object' && target[k] !== null ? applyOverrides(target[k], v) : structuredClone(v);
  }
  return target;
}

attachDom

Browser wiring: keyboard, mouse, touch → Input. toWorld(x,y) converts canvas coords (backend supplies). Returns detach().

Import from nord. Definition from runtime/core/input.js; names and defaults below are taken directly from the runtime.

export function attachDom(input, el, win = window, toWorld = (x, y) => [x, y]) {
  const kd = (e) => { input.setKey(e.code, true); if (e.code === 'Space' || e.code.startsWith('Arrow')) e.preventDefault(); };
  const ku = (e) => input.setKey(e.code, false);
  const pos = (e) => {
    const r = el.getBoundingClientRect();
    const x = ((e.clientX - r.left) / r.width) * (el.width || r.width), y = ((e.clientY - r.top) / r.height) * (el.height || r.height);
    Object.assign(input.pointer, { x, y }); [input.pointer.worldX, input.pointer.worldY] = toWorld(x, y);
  };
  const pd = (e) => { pos(e); input.pointer.down = true; input.setKey(`Mouse${e.button ?? 0}`, true); };
  const pu = (e) => { pos(e); input.pointer.down = false; input.setKey(`Mouse${e.button ?? 0}`, false); };
  const blur = () => input.held.clear();
  win.addEventListener('keydown', kd); win.addEventListener('keyup', ku); win.addEventListener('blur', blur);
  el.addEventListener('pointerdown', pd); win.addEventListener('pointerup', pu); el.addEventListener('pointermove', pos);
  return () => {
    win.removeEventListener('keydown', kd); win.removeEventListener('keyup', ku); win.removeEventListener('blur', blur);
    el.removeEventListener('pointerdown', pd); win.removeEventListener('pointerup', pu); el.removeEventListener('pointermove', pos);
  };
}

virtualJoystick

On-screen joystick (touch/mouse). Drives the four direction actions + input.stick.

Import from nord. Definition from runtime/core/input.js; names and defaults below are taken directly from the runtime.

export function virtualJoystick(input, parent, { left = 'left', right = 'right', up = 'up', down = 'down', size = 120 } = {}) {
  const doc = parent.ownerDocument;
  const base = doc.createElement('div');
  base.style.cssText = `position:absolute;left:24px;bottom:24px;width:${size}px;height:${size}px;border-radius:50%;background:rgba(255,255,255,.12);touch-action:none;user-select:none`;
  const knob = doc.createElement('div');
  knob.style.cssText = `position:absolute;left:50%;top:50%;width:${size / 2}px;height:${size / 2}px;margin:-${size / 4}px;border-radius:50%;background:rgba(255,255,255,.35)`;
  base.append(knob); parent.append(base);
  const set = (x, y) => {
    input.stick = [x, y];
    input.setVirtual(left, x < -0.4); input.setVirtual(right, x > 0.4); input.setVirtual(up, y < -0.4); input.setVirtual(down, y > 0.4);
    knob.style.transform = `translate(${x * size / 3}px,${y * size / 3}px)`;
  };
  const move = (e) => {
    const r = base.getBoundingClientRect();
    let x = (e.clientX - (r.left + r.width / 2)) / (r.width / 2), y = (e.clientY - (r.top + r.height / 2)) / (r.height / 2);
    const l = Math.hypot(x, y); if (l > 1) { x /= l; y /= l; }
    set(x, y);
  };
  base.addEventListener('pointerdown', (e) => { base.setPointerCapture(e.pointerId); move(e); });
  base.addEventListener('pointermove', (e) => { if (base.hasPointerCapture(e.pointerId)) move(e); });
  const end = () => set(0, 0);
  base.addEventListener('pointerup', end); base.addEventListener('pointercancel', end);
  return { el: base, destroy: () => base.remove() };
}

export const ease = {
  linear: (t) => t,
  quadIn: (t) => t * t,
  quadOut: (t) => 1 - (1 - t) * (1 - t),
  quadInOut: (t) => (t < 0.5 ? 2 * t * t : 1 - (-2 * t + 2) ** 2 / 2),
  cubicIn: (t) => t ** 3,
  cubicOut: (t) => 1 - (1 - t) ** 3,
  cubicInOut: (t) => (t < 0.5 ? 4 * t ** 3 : 1 - (-2 * t + 2) ** 3 / 2),
  sineInOut: (t) => -(Math.cos(Math.PI * t) - 1) / 2,
  backOut: (t) => 1 + 2.70158 * (t - 1) ** 3 + 1.70158 * (t - 1) ** 2,
  elasticOut: (t) => (t === 0 || t === 1 ? t : 2 ** (-10 * t) * Math.sin(((t * 10 - 0.75) * (2 * Math.PI)) / 3) + 1),
  bounceOut,
  bounceIn: (t) => 1 - bounceOut(1 - t),
};

touchToStart

Optional web audio entry gate. Consumes the first pointer/key gesture and its release, unlocks game.audio, and leaves the game's current screen untouched. No gate is installed until a game explicitly calls this helper. destroy() removes all UI/listeners.

Import from nord. Definition from runtime/core/start-gate.js; names and defaults below are taken directly from the runtime.

export function touchToStart(game, { parent = globalThis.document?.body, title = 'READY', label = 'TOUCH OR CLICK TO START AUDIO' } = {}) {
  if (!parent?.ownerDocument) return { active: false, destroy() {} };
  const doc = parent.ownerDocument, win = doc.defaultView, el = doc.createElement('div');
  el.dataset.nordStartGate = ''; el.setAttribute('role', 'button'); el.setAttribute('aria-label', label); el.tabIndex = 0;
  el.style.cssText = 'position:fixed;inset:0;z-index:10000;display:grid;place-content:center;text-align:center;background:rgba(3,8,15,.94);color:#c9e9d4;font:14px/2 ui-monospace,monospace;letter-spacing:2px;touch-action:none;user-select:none;cursor:pointer';
  const heading = doc.createElement('strong'), caption = doc.createElement('span'); heading.textContent = title; caption.textContent = label; caption.style.maxWidth = '85vw'; el.append(heading, caption); parent.append(el);
  const types = ['pointerdown', 'pointerup', 'pointercancel', 'touchstart', 'touchend', 'touchcancel', 'click', 'keydown', 'keyup'];
  let active = true, key = null, pointer = null, released = false, timer = null;
  const remove = () => { for (const type of types) win.removeEventListener(type, consume, true); if (timer) win.clearTimeout(timer); el.remove(); };
  const block = e => { if (e.cancelable) e.preventDefault(); e.stopImmediatePropagation(); };
  function consume(e) {
    if (active) {
      if (e.type === 'keydown' && (e.repeat || e.ctrlKey || e.metaKey || e.altKey)) return;
      block(e);
      if (!['pointerdown', 'touchstart', 'keydown'].includes(e.type)) return;
      active = false; key = e.type === 'keydown' ? e.code : null; pointer = e.type === 'keydown' ? null : e.pointerId ?? 'touch';
      el.remove(); game.input?.held?.clear(); game.audio?.unlock();
      return;
    }
    // A distinct new gesture is normal game input, even if a browser omitted the first release/click.
    if (e.type === 'pointerdown' || (e.type === 'touchstart' && released) || (e.type === 'keydown' && (e.code !== key || !e.repeat))) { remove(); return; }
    if ((key && ['keyup', 'keydown'].includes(e.type) && e.code === key) || (pointer !== null && ['pointerup', 'pointercancel', 'touchstart', 'touchend', 'touchcancel', 'click'].includes(e.type))) {
      block(e);
      if (['click', 'pointercancel', 'touchcancel'].includes(e.type)) remove();
      else if (e.type === 'keyup') timer = win.setTimeout(remove, 0);
      else if (e.type === 'pointerup' || e.type === 'touchend') { released = true; game.audio?.unlock(); if (timer) win.clearTimeout(timer); timer = win.setTimeout(remove, 500); }
    }
  }
  for (const type of types) win.addEventListener(type, consume, { capture: true, passive: false });
  return { get active() { return active; }, destroy() { active = false; remove(); } };
}

createGame

@param project parsed nord.project.json @param opts { container: HTMLElement (position:relative), canvas, backend, files, storage, importScript, onError, seed, inlineOnly }

Import from nord. Definition from runtime/core/boot.js; names and defaults below are taken directly from the runtime.

export function createGame(project, opts = {}) {
  const loader = new Loader({ files: opts.files, inlineOnly: opts.inlineOnly });
  const game = new Game(project, {
    backend: opts.backend, storage: opts.storage, importScript: opts.importScript, onError: opts.onError, seed: opts.seed,
    fetchJSON: (p) => loader.json(p),
  });
  game.loader = loader;
  game.audio = new Audio(loader, { saveKey: `nord:${project.id}:audio`, storage: opts.storage ?? globalThis.localStorage });
  if (opts.container) {
    game.ui = new UI(opts.container, { skin: project.ui?.skin, theme: project.ui?.theme, fontBase: `${opts.libBase ?? '/vendor'}/fonts`, onSound: (n) => game.audio.play(`assets/audio/sfx/${n}.wav`, { bus: 'ui' }) });
    game.audio.attach(globalThis.window);
  }
  if (opts.canvas) game._detachInput = attachDom(game.input, opts.canvas, globalThis.window, (x, y) => game.backend?.toWorld?.(x, y) ?? [x, y]);
  game.preload = (paths) => game.backend?.preload?.(paths) ?? Promise.resolve();
  return game;
}

registerCoreComponents

Import from nord. Definition from runtime/core/components.js; names and defaults below are taken directly from the runtime.

export function registerCoreComponents(game) {
  const ui = () => game.ui;
  const pos = (d) => ({ x: d.x ?? 0, y: d.y ?? 0, anchor: d.anchor ?? 'tl', w: d.w, h: d.h, screen: d.screen ?? 'hud' });
  const wrap = (make) => ({
    create(e, d, comp) { comp.h = make(d); comp.h.el.hidden = !e.enabled; },
    update(comp) { comp.h.el.hidden = !comp.entity.enabled; },
    destroy(comp) { comp.h?.remove(); },
    set(comp, k, v) { comp.data[k] = v; comp.h.set?.(v); },
  });
  registerComponent('uiPanel', { ...wrap((d) => ui().panel(pos(d))) });
  registerComponent('uiLabel', { ...wrap((d) => ui().label(d.text ?? '', { ...pos(d), size: d.size, color: d.color })) });
  registerComponent('uiImage', { ...wrap((d) => ui().image(d.src, pos(d))) });
  registerComponent('uiBar', { ...wrap((d) => ui().bar(d.value ?? 1, { ...pos(d), color: d.color })) });
  registerComponent('uiButton', {
    create(e, d, comp) { comp.h = ui().button(d.text ?? 'Button', () => { e.emit('click'); e.callScripts('onClick'); if (d.event) game.events.emit(d.event, e); }, pos(d)); },
    destroy(comp) { comp.h?.remove(); },
    update(comp) { comp.h.el.hidden = !comp.entity.enabled; },
  });

  registerComponent('audioSource', {
    create(e, d, comp) {
      comp.play = (o = {}) => { comp.handle?.stop(); comp.handle = game.audio.play(d.clip, { volume: d.volume ?? 1, loop: !!d.loop, bus: d.bus ?? 'sfx', ...o }); return comp.handle; };
      comp.stop = () => comp.handle?.stop();
      if (d.autoplay) comp.play();
    },
    destroy(comp) { if (comp.data.loop) comp.stop(); },
  });

  registerComponent('audioListener', {});

  // spawner: periodically spawns a prefab at the entity's position. data: { prefab, interval, max, offset:[x,y,z], autostart, jitter }
  registerComponent('spawner', {
    create(e, d, comp) {
      comp.live = new Set(); comp.t = d.interval ?? 1; comp.on = d.autostart !== false;
      comp.spawn = async () => {
        const w = e.worldPos, o = d.offset ?? [0, 0, 0], j = d.jitter ?? [0, 0, 0];
        const r = () => (game.rng.next() - 0.5) * 2;
        const ent = await e.scene.spawn(d.prefab, { pos: [w[0] + o[0] + r() * j[0], w[1] + o[1] + r() * j[1], w[2] + o[2] + r() * j[2]] });
        comp.live.add(ent); return ent;
      };
    },
    update(comp, dt) {
      const d = comp.data;
      for (const x of comp.live) if (x.destroyed) comp.live.delete(x);
      if (!comp.on || !comp.entity.enabled) return;
      comp.t -= dt;
      if (comp.t <= 0) { comp.t += d.interval ?? 1; if (!d.max || comp.live.size < d.max) comp.spawn(); }
    },
  });

  // path: waypoint list with sampling. data: { points:[[x,y,z]...], loop }
  registerComponent('path', {
    create(e, d, comp) {
      const pts = d.points ?? [];
      const segs = []; let total = 0;
      for (let i = 0; i < pts.length - (d.loop ? 0 : 1); i++) {
        const a = pts[i], b = pts[(i + 1) % pts.length];
        const l = Math.hypot(...a.map((v, k) => b[k] - v)); segs.push([a, b, l]); total += l;
      }
      comp.length = total;
      /** Point at distance `s` along the path (wraps if loop, clamps otherwise). */
      comp.at = (s) => {
        if (!segs.length) return pts[0] ? [...pts[0]] : [0, 0, 0];
        s = d.loop ? ((s % total) + total) % total : Math.min(Math.max(s, 0), total);
        for (const [a, b, l] of segs) { if (s <= l || (a === segs.at(-1)[0])) { const t = l ? Math.min(s / l, 1) : 0; return a.map((v, k) => v + (b[k] - v) * t); } s -= l; }
        return [...segs.at(-1)[1]];
      };
      comp.points = pts;
    },
  });
}

fontFaceCSS

@font-face rules for every vendored family (or only families). base is where vendor/fonts is served (dev: '/vendor/fonts').

Import from nord. Definition from runtime/core/themes.js; names and defaults below are taken directly from the runtime.

export function fontFaceCSS(base = '/vendor/fonts', families = Object.keys(FONTS)) {
  return families.map((fam) => {
    const f = FONTS[fam];
    return f.weights.map((w) => `@font-face{font-family:'${fam}';font-weight:${w};font-style:normal;font-display:swap;src:url('${base}/${f.dir}/${f.dir}-latin-${w}-normal.woff2') format('woff2')}`).join('');
  }).join('\n');
}

themeVars

CSS custom properties for a skin+theme pair: --nord-* (UI colors), --nord-font-ui, --nord-font-code, --nord-code-* (syntax).

Import from nord. Definition from runtime/core/themes.js; names and defaults below are taken directly from the runtime.

export function themeVars(skinId = DEFAULTS.skin, themeId = DEFAULTS.theme) {
  const s = skin(skinId), t = theme(themeId), v = {};
  for (const [k, c] of Object.entries(t.ui)) v[`--nord-${k.replace(/[A-Z]/g, (m) => `-${m.toLowerCase()}`)}`] = c;
  for (const [k, c] of Object.entries(t.code)) v[`--nord-code-${k.replace(/[A-Z]/g, (m) => `-${m.toLowerCase()}`)}`] = c;
  v['--nord-font-ui'] = stack(s.ui, 'sans');
  v['--nord-font-code'] = stack(s.code, 'mono');
  v['--nord-code-ligatures'] = s.ligatures ? 'normal' : 'none';
  return v;
}

applyTheme

Inject fonts once and apply vars to el (default: ). Persists the choice. Returns { skin, theme }.

Import from nord. Definition from runtime/core/themes.js; names and defaults below are taken directly from the runtime.

export function applyTheme(el = globalThis.document?.documentElement, { skin: sk, theme: th, base = '/vendor/fonts', storage = globalThis.localStorage } = {}) {
  const read = (k, d) => { try { return storage?.getItem(`nord:${k}`) ?? d; } catch { return d; } };
  const skinId = skin(sk ?? read('skin', DEFAULTS.skin)).id, themeId = theme(th ?? read('theme', DEFAULTS.theme)).id;
  const doc = el.ownerDocument;
  if (!doc.getElementById('nord-fonts')) { const st = doc.createElement('style'); st.id = 'nord-fonts'; st.textContent = fontFaceCSS(base); doc.head.appendChild(st); }
  for (const [k, val] of Object.entries(themeVars(skinId, themeId))) el.style.setProperty(k, val);
  el.style.setProperty('font-family', 'var(--nord-font-ui)');
  try { storage?.setItem('nord:skin', skinId); storage?.setItem('nord:theme', themeId); } catch { /* storage unavailable */ }
  return { skin: skinId, theme: themeId };
}
